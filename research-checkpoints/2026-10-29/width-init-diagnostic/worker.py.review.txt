#!/usr/bin/env python3

HERE=pathlib.Path(__file__).resolve().parent
SRC=pathlib.Path('__REVIEWED_SOURCE_DIRECTORY__')

def sha(p):return hashlib.sha256(pathlib.Path(p).read_bytes()).hexdigest()
def put(p,obj):
 with pathlib.Path(p).open('x') as f:json.dump(obj,f,indent=2,sort_keys=True,allow_nan=False);f.flush();os.fsync(f.fileno())

P=load(HERE/'PROTOCOL.json')
resource.setrlimit(resource.RLIMIT_CPU,(120,121))
signal.signal(signal.SIGTERM,lambda *_: (_ for _ in ()).throw(TimeoutError('watchdog TERM')))

t0=time.monotonic();rows=[]
try:
 for p,h in P['pins'].items():

 def state(module,p):
  d=arrays(p);e=module.state_dict()
  if set(d)!=set(e):raise RuntimeError('state keys')
  for k,a in d.items():
   if a.shape!=tuple(e[k].shape) or a.dtype!=e[k].numpy().dtype or not np.isfinite(a).all():raise RuntimeError('state dtype/shape/finite')
  module.load_state_dict({k:torch.from_numpy(v) for k,v in d.items()});return module
 models={};teachers={};meta={}
 for tid in P['teachers']:
  b=BASE/f'seed_{tid}';old=load(b/'result.json');teacher=m.freeze(state(m.Net(old['seed']['teacher']),b/'teacher_final.npz'))
  if m.state_hash(teacher)!=old['teacher_final_sha256']:raise RuntimeError('teacher identity')
  teachers[tid]=teacher;meta[tid]=old
  a=arrays(b/'fit_targets.npz')
  if set(a)!={'a0','a1','a2'} or any(x.shape!=(1077,64) or x.dtype!=np.float32 or not np.isfinite(x).all() for x in a.values()):raise RuntimeError('targets')

  for r in range(8):

   for budget in (2048,3072):

    put(OUT/f'{key}_START.json',{'teacher':tid,'replicate':r,'seed':seed,'budget':budget,'initial_hash':m.state_hash(mod)})

    m.save_state(OUT/f'{key}_final.npz',mod);models[(tid,r,budget)]=mod
    put(OUT/f'{key}_FIT.json',{'fit':fit,'sqrt_counts':cnt,'final_hash':m.state_hash(mod)})
    if m.state_hash(teacher)!=old['teacher_final_sha256'] or any(p.grad is not None for p in teacher.parameters()):raise RuntimeError('teacher mutated')
 if len(models)!=32:raise RuntimeError('fit census')

 # Only now materialize validation; no validation forward/labels during fitting.

 for tid in P['teachers']:

  if not np.array_equal(split['train'],data['train_ids']) or not np.array_equal(split['validation'],data['validation_ids']):raise RuntimeError('split identity')
