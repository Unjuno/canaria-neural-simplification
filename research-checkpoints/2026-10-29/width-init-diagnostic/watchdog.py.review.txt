#!/usr/bin/env python3
import os,sys,pathlib,json,hashlib,time,signal,subprocess,fcntl
H=pathlib.Path(__file__).resolve().parent
PYTHON='__REVIEWED_PYTHON__'
def sha(p):return hashlib.sha256(pathlib.Path(p).read_bytes()).hexdigest()
def put(p,o):
 with p.open('x') as f:json.dump(o,f,indent=2,sort_keys=True);f.flush();os.fsync(f.fileno())
review=json.loads((H/'INDEPENDENT_GO.json').read_text())
if review.get('decision')!='GO' or not review.get('reviewer') or review.get('files')!={n:sha(H/n) for n in ('worker.py','watchdog.py','PROTOCOL.json')}:raise RuntimeError('missing exact independent GO')
locks=[]
for p in ('__REVIEWED_CPU_LOCK__','__REVIEWED_SUPERVISOR_LOCK__'):
 f=open(p,'a');fcntl.flock(f,fcntl.LOCK_EX|fcntl.LOCK_NB);locks.append(f)
put(H/'CONSUMED_CLAIM.json',{'id':H.name,'utc_epoch':time.time(),'pid':os.getpid(),'review_sha256':sha(H/'INDEPENDENT_GO.json'),'no_retry':True})
t0=time.monotonic();child=None;status='FAILED';reason=None
try:
 env=os.environ.copy();env['PYTHONDONTWRITEBYTECODE']='1'
 with (H/'stdout.log').open('xb') as stdout,(H/'stderr.log').open('xb') as stderr:
  child=subprocess.Popen([PYTHON,'-B',str(H/'worker.py')],stdin=subprocess.DEVNULL,stdout=stdout,stderr=stderr,env=env,start_new_session=True)
  put(H/'START.json',{'pid':child.pid,'pgid':child.pid,'wall_budget_seconds':180,'cpu_budget_seconds':120})
  def interrupted(signum,frame):raise InterruptedError('watchdog signal '+str(signum))
  signal.signal(signal.SIGTERM,interrupted);signal.signal(signal.SIGINT,interrupted)
  try:rc=child.wait(timeout=max(.001,180-(time.monotonic()-t0)))
  except subprocess.TimeoutExpired:raise TimeoutError('wall180seconds')
  terminal=json.loads((H/'output/WORKER_TERMINAL.json').read_text())
  status='COMPLETE' if rc==0 and terminal.get('status')=='COMPLETE' else 'FAILED'
  reason=None if status=='COMPLETE' else {'returncode':rc,'worker':terminal}
except BaseException as e:reason=repr(e)
finally:
 # Cleanup is owned-child-only. Exiting between poll and signal is benign;
 # always wait/reap and attempt the durable terminal even if cleanup errors.
 cleanup_errors=[]
 signal.signal(signal.SIGTERM,signal.SIG_IGN);signal.signal(signal.SIGINT,signal.SIG_IGN)
 try:
  if child is not None:
   if child.poll() is None:
    try:os.killpg(child.pid,signal.SIGTERM)
    except ProcessLookupError:pass
    try:child.wait(timeout=2)
    except subprocess.TimeoutExpired:
     try:os.killpg(child.pid,signal.SIGKILL)
     except ProcessLookupError:pass
     child.wait()
   else:child.wait()
 except BaseException as e:
  cleanup_errors.append(repr(e));status='FAILED'
  if child is not None:
   try:
    if child.poll() is None:
     try:os.killpg(child.pid,signal.SIGKILL)
     except ProcessLookupError:pass
    child.wait(timeout=5)
   except BaseException as e2:cleanup_errors.append(repr(e2))
 finally:
  reaped=child is None or child.poll() is not None
  if not reaped:status='FAILED'
  put(H/'TERMINAL.json',{'status':status,'reason':reason,'cleanup_errors':cleanup_errors,'elapsed':time.monotonic()-t0,'returncode':None if child is None else child.returncode,'child_reaped':reaped,'no_retry':True})
sys.exit(0 if status=='COMPLETE' else 1)
